# Independently authored Windows core-only build. No UI, debugger, packaging, or deployment.
cmake_minimum_required(VERSION 3.24)
project(drawer_diagnostics_toolchain LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Use current x64 MSVC on Windows, or bootstrap_linux.py on Debian/GCC")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(DCONSTRUCT_VENDOR "" CACHE PATH "Clean pinned upstream checkout")
set(CTRE_HEADER_DIR "" CACHE PATH "Folder containing verified ctre.hpp")
execute_process(COMMAND git -C "${DCONSTRUCT_VENDOR}" rev-parse HEAD OUTPUT_VARIABLE PIN OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT PIN STREQUAL "583b38d8b25b2f0b2cd079db70d8caac0a69f3a7")
  message(FATAL_ERROR "Wrong dconstruct commit")
endif()
execute_process(COMMAND git -C "${DCONSTRUCT_VENDOR}" status --porcelain OUTPUT_VARIABLE DIRTY OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(DIRTY)
  message(FATAL_ERROR "Upstream checkout must be clean")
endif()
file(SHA256 "${CTRE_HEADER_DIR}/ctre.hpp" CTRE_SHA)
if(NOT CTRE_SHA STREQUAL "f0d9a70c782914f83066554cc0690e6c68c83f5bfbc8fab111d3f6affdcc653a")
  message(FATAL_ERROR "CTRE hash mismatch")
endif()
set(SEED "${DCONSTRUCT_VENDOR}/test/fixtures/dc/sidbase_new_fixed.bin")
file(SHA256 "${SEED}" SEED_SHA)
if(NOT SEED_SHA STREQUAL "c2d983bfb5afbfc51591a2d9c003e2e801abe841042768107dc0bd91ce5ceb94")
  message(FATAL_ERROR "SID seed hash mismatch")
endif()
set(SOURCE "${DCONSTRUCT_VENDOR}/source")
set(GV "${SOURCE}/Graphviz-13.1.0-win64")
file(GLOB_RECURSE CORE CONFIGURE_DEPENDS "${SOURCE}/ast/*.cpp" "${SOURCE}/base/*.cpp" "${SOURCE}/compilation/*.cpp" "${SOURCE}/decompilation/*.cpp" "${SOURCE}/disassembly/*.cpp")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/buildinfo.h" "#pragma once\n#define VERSION \"2.2.0-pinned-583b38d8\"\n#define COMPILER_VERSION \"MSVC private core-only driver\"\n#define BUILD_DATE \"2026-10-12\"\n")
add_library(core STATIC ${CORE})
target_include_directories(core PUBLIC "${DCONSTRUCT_VENDOR}/include" "${CMAKE_CURRENT_BINARY_DIR}" "${GV}/include" "${CTRE_HEADER_DIR}")
target_compile_definitions(core PUBLIC WIN32 NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(core PUBLIC "/FI${CMAKE_CURRENT_SOURCE_DIR}/src/prelude.h" /templateDepth:8192)
target_link_libraries(core PUBLIC "${GV}/lib/gvc.lib" "${GV}/lib/cgraph.lib" "${GV}/lib/gvplugin_core.lib" "${GV}/lib/gvplugin_pango.lib")
add_executable(compile_core src/compile_core.cpp)
add_executable(dconstruct "${SOURCE}/main_cli.cpp")
foreach(TARGET compile_core dconstruct)
  target_link_libraries(${TARGET} PRIVATE core)
  add_custom_command(TARGET ${TARGET} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${SEED}" "$<TARGET_FILE_DIR:${TARGET}>/sidbase.bin")
  file(GLOB DLLS "${GV}/bin/*.dll")
  add_custom_command(TARGET ${TARGET} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different ${DLLS} "$<TARGET_FILE_DIR:${TARGET}>" COMMAND_EXPAND_LISTS)
endforeach()
